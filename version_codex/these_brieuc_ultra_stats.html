metric_ci <- function(num, den) {
  if (is.na(den) || den == 0) return(c(est = NA_real_, low = NA_real_, high = NA_real_))
  bt <- binom.test(num, den)
  c(est = num / den, low = bt$conf.int[1], high = bt$conf.int[2])
}

lr_ci <- function(tp, fp, fn, tn) {
  a <- tp; b <- fp; c <- fn; d <- tn
  lrp <- if ((1 - d / (b + d)) == 0) Inf else (a / (a + c)) / (1 - d / (b + d))
  lrn <- if ((d / (b + d)) == 0) Inf else (1 - a / (a + c)) / (d / (b + d))
  if (any(c(a, b, c, d) == 0)) {
    a <- a + 0.5; b <- b + 0.5; c <- c + 0.5; d <- d + 0.5
  }
  se_log_lrp <- sqrt((1 / a) - (1 / (a + c)) + (1 / b) - (1 / (b + d)))
  se_log_lrn <- sqrt((1 / c) - (1 / (a + c)) + (1 / d) - (1 / (b + d)))
  lrp_adj <- (a / (a + c)) / (1 - d / (b + d))
  lrn_adj <- (1 - a / (a + c)) / (d / (b + d))
  ci_lrp <- exp(log(lrp_adj) + c(-1, 1) * 1.96 * se_log_lrp)
  ci_lrn <- exp(log(lrn_adj) + c(-1, 1) * 1.96 * se_log_lrn)
  list(
    lrp = c(est = lrp, low = ci_lrp[1], high = ci_lrp[2]),
    lrn = c(est = lrn, low = ci_lrn[1], high = ci_lrn[2])
  )
}

diag_metrics <- function(test, gold, label, doubtful_negative = FALSE) {
  gold2 <- gold
  if (doubtful_negative) gold2[gold2 == "Douteux"] <- "Negatif"
  keep <- !is.na(test) & !is.na(gold2) & gold2 %in% c("Positif", "Negatif")
  test2 <- test[keep]
  gold2 <- gold2[keep]

  tp <- sum(test2 == "Positif" & gold2 == "Positif")
  fn <- sum(test2 == "Negatif" & gold2 == "Positif")
  fp <- sum(test2 == "Positif" & gold2 == "Negatif")
  tn <- sum(test2 == "Negatif" & gold2 == "Negatif")

  se <- metric_ci(tp, tp + fn)
  sp <- metric_ci(tn, tn + fp)
  ppv <- metric_ci(tp, tp + fp)
  npv <- metric_ci(tn, tn + fn)
  lr <- lr_ci(tp, fp, fn, tn)

  roc_obj <- roc(as.numeric(gold2 == "Positif"), as.numeric(test2 == "Positif"), quiet = TRUE)
  auc_ci <- ci.auc(roc_obj)

  data.frame(
    Test = label,
    N = length(test2),
    TP = tp,
    FN = fn,
    FP = fp,
    TN = tn,
    Se = unname(se["est"]),
    Se_low = unname(se["low"]),
    Se_high = unname(se["high"]),
    Sp = unname(sp["est"]),
    Sp_low = unname(sp["low"]),
    Sp_high = unname(sp["high"]),
    VPP = unname(ppv["est"]),
    VPP_low = unname(ppv["low"]),
    VPP_high = unname(ppv["high"]),
    VPN = unname(npv["est"]),
    VPN_low = unname(npv["low"]),
    VPN_high = unname(npv["high"]),
    LRp = unname(lr$lrp["est"]),
    LRp_low = unname(lr$lrp["low"]),
    LRp_high = unname(lr$lrp["high"]),
    LRn = unname(lr$lrn["est"]),
    LRn_low = unname(lr$lrn["low"]),
    LRn_high = unname(lr$lrn["high"]),
    Youden = unname(se["est"] + sp["est"] - 1),
    AUC = as.numeric(auc(roc_obj)),
    AUC_low = as.numeric(auc_ci[1]),
    AUC_high = as.numeric(auc_ci[3]),
    Fisher_p = fisher.test(table(test2, gold2))$p.value,
    stringsAsFactors = FALSE
  )
}

combo_metrics <- function(n_pos, gold, threshold, doubtful_negative = FALSE) {
  test <- ifelse(n_pos >= threshold, "Positif", "Negatif")
  diag_metrics(test, gold, paste0(">=", threshold, " positifs"), doubtful_negative = doubtful_negative)
}

kappa_summary <- function(a, b, label) {
  keep <- !is.na(a) & !is.na(b)
  k <- psych::cohen.kappa(cbind(a[keep], b[keep]))
  agree <- mean(a[keep] == b[keep])
  data.frame(
    Test = label,
    N = sum(keep),
    Accord = agree,
    Kappa = k$kappa,
    Kappa_low = k$confid["unweighted kappa", "lower"],
    Kappa_high = k$confid["unweighted kappa", "upper"],
    Interpretation = case_when(
      k$kappa < 0.20 ~ "faible",
      k$kappa < 0.40 ~ "passable",
      k$kappa < 0.60 ~ "moderee",
      k$kappa < 0.80 ~ "substantielle",
      TRUE ~ "presque parfaite"
    ),
    stringsAsFactors = FALSE
  )
}

cp_lower_expected <- function(n_neg, spec, alpha = 0.05) {
  qbeta(alpha, n_neg * spec, n_neg * (1 - spec) + 1)
}

required_negatives_cp <- function(spec, width, alpha = 0.05, max_n = 5000) {
  for (n_neg in 2:max_n) {
    lower <- cp_lower_expected(n_neg, spec, alpha = alpha)
    if ((spec - lower) <= width) return(n_neg)
  }
  NA_integer_
}

3 Méthodes

Les analyses ont été réalisées sous R (R Foundation for Statistical Computing, Vienne, Autriche) à partir du fichier recueil_bon.csv, après harmonisation des modalités textuelles et recodage des variables cliniques en indicateurs binaires lorsque nécessaire. Conformément au choix méthodologique retenu pour ce rapport, tous les cas de gold standard “Douteux” ont été reclassés comme “Négatif” et cette règle a été appliquée de façon homogène à l’ensemble des analyses principales. Les variables démographiques et cliniques ont d’abord été décrites par des statistiques descriptives adaptées à leur nature. Les performances diagnostiques des tests U1, U2a, U2b, U3 et Scratch Collapse Test ont ensuite été évaluées par construction des tableaux de contingence 2x2, avec estimation des effectifs TP, FP, FN et TN, de la sensibilité, de la spécificité, des valeurs prédictives positive et négative, des rapports de vraisemblance positif et négatif, de l’indice de Youden et de leurs intervalles de confiance exacts à 95% lorsque pertinent. Les combinaisons de tests ULNT ont été étudiées selon plusieurs seuils de positivité (>=1, >=2, >=3 et 4/4 tests positifs). La reproductibilité inter-observateur entre évaluateurs 1 et 2 a été quantifiée par le coefficient kappa de Cohen avec intervalle de confiance. Les courbes ROC et les aires sous la courbe ont été calculées à titre de résumé descriptif complémentaire, tout en considérant que la nature binaire des prédicteurs limite leur apport par rapport aux indices diagnostiques classiques. Les résultats sont présentés dans l’ordre suivant, comme dans un manuscrit scientifique: description de la population et préparation des données, justification de l’effectif, performances diagnostiques avec tableaux de contingence, précision des estimations, analyses de sensibilité, combinaisons de tests, analyses descriptives complémentaires (niveaux métamériques, corrélations, reproductibilité), probabilités post-test et enfin discussion critique de la validité statistique.

4 Justification de l’effectif

Les captures transmises documentent un calcul de taille d’échantillon centré sur la précision de la spécificité par intervalle de confiance exact de Clopper-Pearson. Ici, les scénarios ne sont pas seulement recopiés: ils sont recalculés dans R à partir d’une borne inférieure unilatérale exacte à 95%, en recherchant le plus petit nombre de sujets négatifs permettant d’obtenir une largeur cible ω telle que:

[ = t - L_{95%, exact} ]

où t est la spécificité cible et L la borne inférieure exacte de Clopper-Pearson. En pratique, ω n’est pas la largeur totale d’un intervalle bilatéral: c’est la marge de sécurité vers le bas, c’est-à-dire l’écart maximal toléré entre la spécificité visée et sa borne inférieure. Par exemple, si l’on vise une spécificité de 0,90 avec ω = 0,10, on veut garantir que la borne inférieure de l’IC exact reste autour de 0,80 ou plus. Plus ω est petit, plus on exige une estimation précise, donc plus l’effectif nécessaire augmente.

L’effectif total requis est ensuite obtenu en tenant compte de la prévalence attendue:

[ N_{total} = ]

Cette logique est celle de l’image fournie.

En pratique, le recalcul retrouve les ordres de grandeur de l’image: il faut environ 50 négatifs pour viser une spécificité de 0,90 avec une largeur proche de 0,10, contre 59 négatifs pour une spécificité de 0,85 avec la même exigence de précision.

À quoi sert concrètement cette justification d’effectif ? - Elle sert à juger si l’étude est capable d’estimer la spécificité avec une précision acceptable, et pas seulement à produire une valeur ponctuelle. - Elle montre que, dans cette cohorte, le point limitant est surtout le nombre de sujets négatifs, puisque la spécificité est calculée sur eux. - Elle permet donc de dire, en toutes lettres, que l’étude actuelle peut soutenir une première estimation de la spécificité, mais pas une estimation très précise pour tous les tests. - Elle aide enfin à planifier une cohorte de confirmation: si l’objectif est une spécificité autour de 0,90 avec une bonne borne inférieure, il faudra surtout augmenter le nombre de sujets non atteints, donc le nombre total inclus selon la prévalence attendue.

On peut réintégrer ici le NNT exploratoire mentionné par la statisticienne, à condition d’être très clair sur son statut.

Définition utilisée ici. On définit un gain absolu exploratoire de probabilité diagnostique comme:

[ ARR_{exploratoire} = VPP - ]

et donc:

[ NNT_{exploratoire} = ]

Autrement dit, cet indicateur cherche à traduire en termes cliniques combien de tests positifs “sont nécessaires” pour obtenir un diagnostic positif additionnel par rapport à la probabilité pré-test de la cohorte.

Interprétation. - Plus le NNT exploratoire est bas, plus un test positif apporte un gain de probabilité diagnostique au-delà de la prévalence initiale. - Dans cette cohorte, les meilleurs profils exploratoires sont U1 et U2b, ce qui est cohérent avec les autres résultats du rapport.

Limites. - Ce NNT n’est pas un indicateur standard de validité diagnostique. - Il dépend fortement de la prévalence de la cohorte. - Il ne doit pas remplacer Se/Sp/LR/VPP/VPN dans la conclusion scientifique. - Il est donc présenté comme un complément clinique exploratoire, pas comme un résultat principal.

Bilan critique. - Le choix de traiter les 3 gold standards Douteux comme Negatif doit être annoncé clairement, car il influence certaines estimations et peut modifier légèrement les AUC et les indices dérivés. - L’analyse de sensibilité par exclusion des Douteux montre toutefois que le signal principal (U1/U2b utiles, U3 faible) reste globalement stable. - La section par niveau métamérique est incomplète et trop sparse pour soutenir autre chose qu’une description. - Les AUC sont acceptables comme résumé, mais secondaires par rapport aux indices diagnostiques classiques, car les tests sont binaires. - Le “NNT” ne doit pas faire partie de la conclusion scientifique d’une étude diagnostique; il a donc été remplacé par des probabilités post-test. - Les conclusions cliniques restent fragiles en spécificité, car il n’y a que 14 négatifs dans l’analyse principale; cela élargit fortement les IC des spécificités et des LR, surtout pour Scratch.

15 Conclusion intégrée

Dans cette cohorte, U1 et U2b sont les tests isolés les plus utiles. U1 est le plus sensible et U2b offre le meilleur compromis global. Scratch est surtout un test de confirmation quand il est positif, mais sa faible sensibilité l’empêche d’être un bon test d’exclusion. U3 apparaît peu performant.

Pour les combinaisons, au moins un ULNT positif est la meilleure stratégie de dépistage et au moins deux ULNT positifs offre le meilleur équilibre pratique. En revanche, les conclusions par niveau métamérique sont trop peu robustes pour être retenues analytiquement.

Sur le plan méthodologique, le document est désormais plus valide statistiquement parce qu’il: - sépare analyse principale et analyse de sensibilité pour les Douteux; - rapporte les résultats extraits et leurs intervalles de confiance; - cadre le NNT comme indicateur exploratoire distinct du noyau principal; - explicite les limites dues aux effectifs, aux données manquantes et à la nature binaire des tests.
